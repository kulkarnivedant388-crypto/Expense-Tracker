// Core C++ Engine Logic Underlying the Web Application Data Flow

#include <iostream>
#include <vector>
#include <string>
#include <fstream>
#include <sstream>
#include <numeric>

struct Expense {
    double amount;
    std::string category;
    std::string date;
    std::string description;

    std::string toCSV() const {
        return std::to_string(amount) + "," + category + "," + date + "," + description;
    }
};

class ExpenseTracker {
private:
    std::vector<Expense> expenses;
    const std::string filename = "expenses_data.csv";
public:
    void addExpense(double amt, std::string cat, std::string dt, std::string desc) {
        expenses.push_back({amt, cat, dt, desc});
    }
};

Interview Architectural Highlights:

  • OOP & Encapsulation: Custom `struct Expense` paired with `ExpenseTracker` class.
  • Memory Management: Uses `std::vector<Expense>` for dynamic array resizing without memory leaks.
  • Persistence Layer: Uses C++ file streams (`<fstream>`) for local CSV record saving.
